function escapeHtml(value){ return String(value ?? "").replace(/&/g,"&amp;").replace(/</g,"&lt;").replace(/>/g,"&gt;").replace(/"/g,"&quot;").replace(/'/g,"&#039;"); } 
function escapeAttr(value){ return escapeHtml(value); } 
function escapeJs(value){ return String(value ?? "").replace(/\\/g,"\\\\").replace(/'/g,"\\'").replace(/\n/g,"\\n").replace(/\r/g,""); } 
function normalize(value){ return String(value ?? "").trim().toLowerCase(); } 
function generateId(){ return Date.now().toString(36) + Math.random().toString(36).substring(2,8); }

function getTourDateTime(){ 
  const now = new Date(); 
  const day = String(now.getDate()).padStart(2,"0"); 
  const month = String(now.getMonth()+1).padStart(2,"0"); 
  const year = now.getFullYear(); 
  const hours = String(now.getHours()).padStart(2,"0"); 
  const minutes = String(now.getMinutes()).padStart(2,"0"); 
  return day+"/"+month+"/"+year+" "+hours+":"+minutes; 
}

function ensureRouteData(vehicle){ 
  if(!vehicle.route){ vehicle.route = { driver:"", from:"", to:"", status:"" }; } 
  if(typeof vehicle.route.driver !== "string"){ vehicle.route.driver=""; } 
  if(typeof vehicle.route.from !== "string"){ vehicle.route.from=""; } 
  if(typeof vehicle.route.to !== "string"){ vehicle.route.to=""; } 
  if(typeof vehicle.route.status !== "string"){ vehicle.route.status=""; } 
  return vehicle.route; 
}

function addTourToHistory(vehicle){ 
  if(!vehicle){ return; } 
  const route=ensureRouteData(vehicle); 
  if( !route.driver || !route.from || !route.to ){ return; } 
  if(route.historyId){ 
    const alreadyExists= tourHistory.some( item=>item.id===route.historyId ); 
    if(alreadyExists){ return; } 
  } 
  const historyId=generateId(); 
  const historyItem={ id:historyId, date:getTourDateTime(), vehicleId:vehicle.id, model:vehicle.model, variant:vehicle.variant, color:vehicle.color, chassis:vehicle.chassis, from:route.from, to:route.to, driver:route.driver }; 
  tourHistory.unshift(historyItem); 
  route.historyId=historyId; 
  saveTourHistory(); 
  saveVehicles(); 
}

function getCategoryVehicles(category){ 
  if(category==="Total"){ 
    return vehicles.filter( v=>v.status!=="Delivered" );

} 

function searchMatchesVehicle( vehicle, searchText ){ 
  if(!searchText){ return true; } 
  const search= normalize(searchText); 
  const searchableText=[ vehicle.model, vehicle.variant, vehicle.color, vehicle.chassis, vehicle.engine, vehicle.key, vehicle.location, vehicle.status ].map(normalize).join(" "); 
  return searchableText.includes(search); 
}

}

  const vehicle= vehicles.find( v=>v.id===id );

  if(vehicle.status!=="Delivered"){ 
    showToast( "Only Delivered vehicles can be deleted" );

  }

function createRoute(){ 
  const driver= document.getElementById( "routeDriver" ).value.trim(); 
  const vehicleId= document.getElementById( "routeVehicle" ).value; 
  const from= document.getElementById( "routeFrom" ).value.trim(); 
  const to= document.getElementById( "routeTo" ).value.trim(); 
  if( !driver || !vehicleId || !from || !to ){ showToast( "Please fill driver, vehicle, From and To" ); return; }

  route.to=to;

  route.historyId=""; 
  vehicle.location=from; 
  saveVehicles(); 
  renderDashboard(); 
  clearRouteForm(); 
  populateRouteVehicles(); 
  renderRoutes(); 
  showToast( "Route started — vehicle location updated to From" ); 
}

function routeMatchesSearch( vehicle, search ){ 
  if(!search){ return true; } 
  const route= ensureRouteData(vehicle); 
  const text=[ vehicle.model, vehicle.variant, vehicle.color, vehicle.chassis, vehicle.engine, vehicle.location, route.driver, route.from, route.to, route.status ].map(normalize).join(" "); 
  return text.includes( normalize(search) ); 
} 

function renderRoutes(){

  const routeVehicles= vehicles.filter( vehicle=> ensureRouteData(vehicle).status );

  document.getElementById( "routeCountBadge" ).textContent= filtered.length+ " / "+ routeVehicles.length; 
  document.getElementById( "routeSearchClear" ).classList.toggle( "show", !!search );

function routeGoing(id){ 
  const vehicle= vehicles.find( v=>v.id===id ); 
  if(!vehicle){ return; } 
  const route= ensureRouteData(vehicle); 
  if( !route.driver || !route.from || !route.to ){ showToast( "Route details are incomplete" ); return; } 
  route.status="Going"; 
  route.historyId=""; 
  vehicle.location=route.from; 
  saveVehicles(); 
  renderDashboard(); 
  renderRoutes(); 
  populateRouteVehicles(); 
  showToast( "GOING — Location updated to From" ); 
} 

function routeReached(id){ 
  const vehicle= vehicles.find( v=>v.id===id ); 
  if(!vehicle){ return; } 
  const route= ensureRouteData(vehicle); 
  if(route.status!=="Going"){ showToast( "Vehicle must be Going first" ); return; } 
  vehicle.location=route.to; 
  route.status="Reached"; 
  saveVehicles(); 
  addTourToHistory(vehicle); 
  renderDashboard(); 
  renderRoutes(); 
  populateRouteVehicles(); 
  showToast( "REACHED — Location updated to "+ route.to ); 
}

function addVehicle(){ 
  if(getCurrentUserRole() !== "admin") return;
  const model= document.getElementById( "addModel" ).value.trim(); 
  const variant= document.getElementById( "addVariant" ).value.trim(); 
  const color= document.getElementById( "addColor" ).value.trim(); 
  const chassis= document.getElementById( "addChassis" ).value.trim(); 
  const engine= document.getElementById( "addEngine" ).value.trim(); 
  const key= document.getElementById( "addKey" ).value.trim(); 
  const location= document.getElementById( "addLocation" ).value.trim(); 
  const status= document.getElementById( "addStatus" ).value; 
  if( !model || !variant || !color || !chassis || !engine || !key || !location ){ showToast( "Please fill all vehicle details" ); return; } 
  const duplicateChassis= vehicles.some( v=> normalize(v.chassis)=== normalize(chassis) ); 
  const duplicateEngine= vehicles.some( v=> normalize(v.engine)=== normalize(engine) ); 
  const duplicateKey= vehicles.some( v=> normalize(v.key)=== normalize(key) ); 
  if(duplicateChassis){ showToast( "Chassis number already exists" ); return; } 
  if(duplicateEngine){ showToast( "Engine number already exists" ); return; } 
  if(duplicateKey){ showToast( "Key number already exists" ); return; } 
  const vehicle={ id:generateId(), model:model, variant:variant, color:color, chassis:chassis, engine:engine, key:key, location:location, status:status, route:{ driver:"", from:"", to:"", status:"" } }; 
  vehicles.push(vehicle); 
  saveVehicles(); 
  renderDashboard(); 
  closeModal( "addModal" ); 
  showToast( "Vehicle added successfully" ); 
}
